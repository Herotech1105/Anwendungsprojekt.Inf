# CA with own CN
openssl genrsa -out ca.key 2048

openssl req -new -x509 -days 3650 -key ca.key -out ca.crt \
  -subj "/CN=mqtt.local"

# Broker
openssl genrsa -out mqtt.key 2048

openssl req -new -key mqtt.key -out mqtt.csr \
  -subj "/CN=192.168.4.18"

openssl x509 -req -days 3650 -in mqtt.csr \
  -CA ca.crt -CAkey ca.key -CAcreateserial -out mqtt.crt

# Check
openssl verify -CAfile ca.crt mqtt.crt

### SAN for CA

# 1. Create config file (san.cnf)
[req]
distinguished_name = req_distinguished_name
req_extensions = v3_req
prompt = no

[req_distinguished_name]
CN = 192.168.4.18

[v3_req]
subjectAltName = @alt_names

[alt_names]
IP.1 = 192.168.4.18
DNS.1 = mqtt
DNS.2 = mosquitto
DNS.3 = localhost
DNS.4 = mqtt.local

# 2. Create new CSR
openssl req -new -key mqtt.key -out mqtt.csr -config san.cnf

# 3. Sign certificate with SAN
openssl x509 -req -days 3650 -in mqtt.csr \
  -CA ca.crt -CAkey ca.key -CAcreateserial \
  -out mqtt.crt -extensions v3_req -extfile san.cnf

# 4. Check
openssl verify -CAfile ca.crt mqtt.crt

openssl x509 -in mqtt.crt -noout -text | grep -A3 "Subject Alternative"
